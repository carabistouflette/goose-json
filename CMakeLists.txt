cmake_minimum_required(VERSION 3.15)
project(goose_json_native LANGUAGES CXX)
add_library(goose_json_native STATIC native/accelerated.cpp native/vendor/simdutf/simdutf.cpp native/vendor/dragonbox/dragonbox_to_chars.cpp)
target_compile_features(goose_json_native PRIVATE cxx_std_17)
target_include_directories(goose_json_native PRIVATE native/vendor/simdutf native/vendor)
set_target_properties(goose_json_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Keep vendored C++ symbols private to this package, even if another package
# bundles a different simdutf version.
target_compile_definitions(goose_json_native PRIVATE simdutf=goose_json_simdutf JKJ_NAMESPACE=goose_json_dragonbox)
# Goose links the program with a C driver, so publish the C++ driver's runtime.
target_link_libraries(goose_json_native PUBLIC ${CMAKE_CXX_IMPLICIT_LINK_LIBRARIES})

option(GOOSE_JSON_NATIVE_TESTS "Build standalone native sanitizer tests" OFF)
option(GOOSE_JSON_FUZZ "Build the Clang libFuzzer harness" OFF)
if(GOOSE_JSON_NATIVE_TESTS OR GOOSE_JSON_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Native sanitizer checks require GCC or Clang")
    endif()
    set(sanitizer_flags -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
    target_compile_options(goose_json_native PRIVATE ${sanitizer_flags})
    target_link_options(goose_json_native INTERFACE -fsanitize=address,undefined)
endif()
if(GOOSE_JSON_NATIVE_TESTS)
    add_executable(goose_json_native_test tests/native.cpp)
    target_compile_features(goose_json_native_test PRIVATE cxx_std_17)
    target_compile_options(goose_json_native_test PRIVATE ${sanitizer_flags} -UNDEBUG)
    target_link_libraries(goose_json_native_test PRIVATE goose_json_native)
endif()
if(GOOSE_JSON_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "libFuzzer requires Clang")
    endif()
    target_compile_options(goose_json_native PRIVATE -fsanitize=fuzzer-no-link)
    target_link_options(goose_json_native INTERFACE -fsanitize=fuzzer-no-link)
    add_executable(goose_json_fuzz tests/fuzz_native.cpp)
    target_compile_features(goose_json_fuzz PRIVATE cxx_std_17)
    target_compile_options(goose_json_fuzz PRIVATE ${sanitizer_flags} -fsanitize=fuzzer)
    target_link_options(goose_json_fuzz PRIVATE -fsanitize=fuzzer)
    target_link_libraries(goose_json_fuzz PRIVATE goose_json_native)
endif()
